================================================================================
                    CORE SWITCH 2 - STANDBY FOR ALL VLANs
                         Simple Redundancy (No Load Balancing)
================================================================================

enable
configure terminal

! ===== HOSTNAME =====
hostname CORE2

! ===== VLAN CREATION =====
vlan 101
  name MGMT
exit

vlan 102
  name Camera
exit

vlan 103
  name WorkStation
exit

vlan 104
  name Server
exit

! ===== SPANNING TREE CONFIGURATION =====
spanning-tree mode rapid-pvst

! CORE2 को सभी VLANs के लिए Secondary (Priority: 8192)
spanning-tree vlan 101 priority 8192
spanning-tree vlan 102 priority 8192
spanning-tree vlan 103 priority 8192
spanning-tree vlan 104 priority 8192

! ===== INTERFACE CONFIGURATIONS =====

! CORE Link (ISL) - CORE2 to CORE1
interface ethernet 0/1
  description CORE-LINK-to-CORE1
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk allowed vlan 101,102,103,104
  no shutdown
exit

! Access Switch 1 Link
interface ethernet 0/3
  description ACCESS1-Link-Secondary
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk allowed vlan 101,102,103,104
  spanning-tree portfast trunk
  no shutdown
exit

! Access Switch 2 Link
interface ethernet 0/2
  description ACCESS2-Link-Secondary
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk allowed vlan 101,102,103,104
  spanning-tree portfast trunk
  no shutdown
exit

! Field Switch Link
interface ethernet 0/0
  description FIELD-Switch-Link
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk allowed vlan 101,102,103,104
  spanning-tree portfast trunk
  no shutdown
exit

! ===== SVI (VLAN INTERFACES) =====

interface vlan 101
  ip address 100.77.191.2 255.255.255.128
  description MGMT-VLAN-SVI
  no shutdown
exit

interface vlan 102
  ip address 100.76.60.11 255.255.254.0
  description Camera-VLAN-SVI
  no shutdown
exit

interface vlan 103
  ip address 100.77.189.71 255.255.255.192
  description WorkStation-VLAN-SVI
  no shutdown
exit

interface vlan 104
  ip address 100.77.189.11 255.255.255.224
  description Server-VLAN-SVI
  no shutdown
exit

! ===== 🟢 HSRP CONFIGURATION - CORE2 STANDBY सभी VLANs में =====

! VLAN 101 - MGMT (CORE2 STANDBY)
interface vlan 101
  standby 1 ip 100.77.191.5
  standby 1 priority 100              ⭐ LOW PRIORITY = STANDBY
  standby 1 preempt
  standby 1 authentication md5 key-string MySecureKey123
  no shutdown
exit

! VLAN 102 - Camera (CORE2 STANDBY)
interface vlan 102
  standby 2 ip 100.76.60.1
  standby 2 priority 100              ⭐ LOW PRIORITY = STANDBY
  standby 2 preempt
  standby 2 authentication md5 key-string MySecureKey123
  no shutdown
exit

! VLAN 103 - WorkStation (CORE2 STANDBY)
interface vlan 103
  standby 3 ip 100.77.189.65
  standby 3 priority 100              ⭐ LOW PRIORITY = STANDBY
  standby 3 preempt
  standby 3 authentication md5 key-string MySecureKey123
  no shutdown
exit

! VLAN 104 - Server (CORE2 STANDBY)
interface vlan 104
  standby 4 ip 100.77.189.1
  standby 4 priority 100              ⭐ LOW PRIORITY = STANDBY
  standby 4 preempt
  standby 4 authentication md5 key-string MySecureKey123
  no shutdown
exit

! ===== OSPF ROUTING =====
router ospf 1
  router-id 100.77.191.2
  network 100.77.191.0 0.0.0.127 area 0
  network 100.76.60.0 0.0.1.255 area 0
  network 100.77.189.0 0.0.0.255 area 0
  passive-interface default
  no passive-interface vlan 101
  no passive-interface vlan 102
  no passive-interface vlan 103
  no passive-interface vlan 104
  auto-cost reference-bandwidth 100000
exit

ip routing

! ===== MANAGEMENT =====
username admin privilege 15 secret admin123
enable secret admin123
service password-encryption

ip http server
ip http secure-server
ip http authentication local

snmp-server community public RO
snmp-server community private RW
snmp-server location "EVE-NG LAB - CORE2"
snmp-server contact "Network Admin"

end
write memory
